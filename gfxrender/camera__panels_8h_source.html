11// Given, GfxRender onward: the same camera toolbar/floating-panel
12// controls shared by every assignment from Camera & Shading (4) through
13// Ray (6) that needs them, rather than each keeping its own copy.
14// Self-contained, like every other panel in this codebase -- Draw()
15// touches only this file's own classes' members, never Camera directly.
16// Interactive navigation (Camera::Move()/Look()/Zoom()/FaceTarget())
17// stays each assignment's own app code, which alone knows Camera's fixed
18// API; direct-editing sync (CameraParametersPanel's own position/look-
19// vector/up-vector/FOV/aspect/clip-plane fields) goes through GfxCamera's
20// own SyncCameraParametersPanelFromCamera()/
21// ApplyCameraParametersPanelToCamera() (camera_gui_sync.h) instead of
22// being hand-duplicated as identical app code in Camera & Shading (4)/
23// iModel (5)/Ray (6) alike. Zero dependency on GfxMath/Camera itself
24// despite the name -- every field below is a plain float/enum -- but
25// lives in GfxRender rather than GfxCore since it's only actually needed
26// from Camera & Shading (4) on, once a live Camera first appears.
27//
28// Camera is a first-person "Minecraft style" fly camera, so
29// CameraControlModePanel's own kFly/kOrbit choice (below) picks between
30// that and a simpler fixed-target orbit, rather than exposing a Dolly/
31// Pan/Zoom/Rotate-style orbit-around-target camera model as the only
32// option.

39// "Preset requested this frame" -- kNone unless a preset button was just

82// Direct-editing parameter display -- owns the actual position/look
83// vector/up vector/height angle/aspect ratio/near/far values shown and
84// dragged in the panel, the classic eye/target-direction/up/fov/aspect/
85// clip camera model Camera's own doc comment describes (rather than
86// this particular camera's internal yaw/pitch representation --
87// ImodelApp is the one place that converts between the two, via
88// Camera::Forward()/set_forward(), the same as Camera & Shading (4)'s
89// own CameraApp did). "Height angle" here is that model's fov, named for
90// what it actually measures -- see Camera::set_fov_y_degrees()'s own doc
91// comment: the value maps straight onto the near plane's *height*, so
92// "height angle" is the more literal name than the more common but
93// ambiguous "FOV" (which, depending on the API, sometimes means the
94// horizontal angle instead). A plain value holder, the same character as
95// every other panel's fields; no behavior of its own -- ImodelApp syncs
96// these from camera_ every frame before Draw() (so edits start from the
97// camera's true current state) and applies them back to camera_ every
98// frame after (so every edit takes effect immediately, live, with no
99// separate Apply button -- unlike MaterialPanel's own explicit-only
100// Apply/Get, direct numeric camera parameters are meant to be tweaked
101// and see the result right away). The accessors return non-const
102// pointers for exactly that sync/apply round trip; every field here is a
103// real, independent, editable parameter, including two that might look
104// like they shouldn't be: up_vector() needn't stay perpendicular to
105// look_vector() (it's fed straight into Camera::set_up_vector(), the
106// "up" hint Matrix4::LookAt() re-orthogonalizes internally, so tilting
107// it introduces roll rather than being rejected or silently
108// re-normalized here), and aspect_ratio() doesn't just mirror the
109// window's own shape -- ImodelApp renders into a letterboxed/pillarboxed
110// viewport matching it instead (see its own ComputeLetterboxViewport()),
111// the same as Camera & Shading (4)'s own CameraApp does, so it's a real,
112// independently editable value rather than one silently overwritten
113// from the real framebuffer shape every frame; ImodelApp::InitOpenGL()
114// seeds it to match the actual window once at startup (so the default
115// view still fills the window edge to edge, no visible margin) without
116// re-syncing it there every frame after, which is what would make
117// editing it here pointless.
118//
119// DrawToolbarButton()/DrawFloatingWindow() (a real floating ImGui window
120// toggled by a toolbar button, the same pattern MaterialPanel's own
121// popped-out Material window uses) beats a transient dropdown popup
122// here: this panel's own values change out from under it every frame
123// the camera moves via WASD/mouse-look, so it needs to stay open and
124// keep refreshing while the user flies around, not close the moment
125// they click away from it into the 3D view.
126class CameraParametersPanel : private ToggleWindowPanelBase {
127 public:
128 // Content only, no window of its own -- called by DrawFloatingWindow()
129 // below.
130 void Draw();
131
132 // A single toolbar button (see tool_button.h/ToggleWindowPanelBase)
133 // that toggles DrawFloatingWindow() below open/closed, highlighted
134 // (the same way ObjectModePanel's own selected mode button is)
135 // whenever that window is currently open. force_open forces the
136 // window open (for headless verification -- see ImodelApp::
137 // ParseCommandLineArg()'s own --open-popup).
138 void DrawToolbarButton(bool force_open = false);
139
140 // Draws Draw()'s own content inside a real, independent, movable
141 // ImGui window if the window is currently open (a no-op otherwise),
142 // rather than every other toolbar dropdown's transient popup.
143 void DrawFloatingWindow();
144
145 float *position() { return position_; }
146 float *look_vector() { return look_vector_; }
147 float *up_vector() { return up_vector_; }
148 float *height_angle_deg() { return &height_angle_deg_; }
149 float *aspect_ratio() { return &aspect_ratio_; }
150 float *near_plane() { return &near_plane_; }
151 float *far_plane() { return &far_plane_; }
152
153 private:
154 float position_[3] = {0.0f, 0.0f, 4.0f};
155 float look_vector_[3] = {0.0f, 0.0f, -1.0f};
156 float up_vector_[3] = {0.0f, 1.0f, 0.0f};
157 float height_angle_deg_ = 60.0f;
158 float aspect_ratio_ = 4.0f / 3.0f;
159 float near_plane_ = 0.01f;
160 float far_plane_ = 100.0f;
161};
