36 /* Looks for a file named basename in each of the paths specified. If found,
37 the full path to the file is returned. If not found, then basename is returned.
38 Example:
39 ~~~
40 std::vector<std::string> search_path;
41 search_path.push_back(".");
42 search_path.push_back("./data");
43 search_path.push_back("./shaders");
44 search_path.push_back("/usr/local/share/blah/blah/data");
45
46 std::string file = Platform::findFile("mydata.csv", search_path);
47 ~~~
48 */
49 static std::string FindFile(const std::string &basename,
50 const std::vector<std::string> &searchpath);
51
52 /* Same search as above, but also records every full candidate path
53 actually checked into *tried, in order, regardless of whether any of
54 them succeeded (when tried is non-null) -- lets a caller whose own
55 load then fails report exactly where it looked, rather than just
56 repeating basename back with no clue why it didn't work.
57 */
58 static std::string FindFile(const std::string &basename,
59 const std::vector<std::string> &searchpath,
60 std::vector<std::string> *tried);
