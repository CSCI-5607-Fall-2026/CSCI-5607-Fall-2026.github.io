11// Given, GfxRender onward: the same Load Scene/Load Obj/Save Scene
12// controls used by every assignment from Camera & Shading (4) through
13// Ray (6) that needs them. "Model or Scene" -- a single path field
14// plus a one-shot "clicked this frame" flag, the same character as
15// MaterialPanel's own texture field -- each assignment's own app class
16// (which alone knows how to load and hold its own live scene) does the
17// actual loading when it sees the flag set. Save (see SaveScenePanel
18// below) is its own separate class with its own field/button, the same
19// shape as this one, rather than a third Kind of this class -- it
20// doesn't share ScenePanel's own "same field, two differently-labeled
21// toolbar buttons" role (there's only ever one Save Scene button) and
22// defaults to its own filename, not "primitives.scene".

29// Direct-editing path field plus a one-shot "clicked this frame" flag.
30// ImodelApp keeps two separate instances of this exact class -- one
31// behind the scene toolbar's own "Load Scene" dropdown button, one
32// behind its "Load Obj" button (see DrawToolbarButton()'s Kind param and
33// ImodelApp::DrawGUI()) -- both still ultimately go through the same
34// ImodelApp::LoadSceneFile(), which tells a real GfxRender SceneIO
35// .scene file apart from a bare Wavefront .obj (wrapped in an ad hoc
36// scene, a same-named .mtl sitting next to it picked up automatically)
37// by extension; the two toolbar buttons exist for discoverability (two
38// clearly-labeled dropdowns instead of one field that silently accepts
39// either), not because the panel or the loading logic itself cares
40// which button triggered it. No New here -- BuildDefaultScene() (called
41// once from InitOpenGL()) is the only way back to a hardcoded starting
42// scene.
43class ScenePanel {
44 public:
45 // Which of the two toolbar roles DrawToolbarButton() below is
46 // filling this call -- an app owning two separate ScenePanel
47 // instances (one for "Load Scene", one for "Load Obj") only ever
48 // differ in icon/tooltip/default path, never in behavior, so one
49 // class fills both roles rather than two near-identical ones.
50 enum Kind { kScene, kObj };
51
52 ScenePanel() = default;
53 explicit ScenePanel(const char *default_path) {
54 SetScenePath(default_path);
55 }
56
57 void Draw();
58
59 // A single toolbar button (see tool_button.h) that reveals this
60 // panel's own Draw() in a dropdown popup anchored right below
61 // itself when clicked.
62 void DrawToolbarButton(Kind kind, bool force_open = false);
63
64 char *scene_path() { return scene_path_; }
65 const char *scene_path() const { return scene_path_; }
66 bool load_requested() const { return load_requested_; }
67 // Overwrites the displayed path -- used after a --load-scene command
68 // line argument loads a different model/scene than whatever this
69 // field currently shows, so the field always shows what's actually
70 // loaded. Truncates to fit if path is longer than the field can
71 // hold.
72 void SetScenePath(const std::string &path);
73
74 private:
75 // Set from DrawToolbarButton()'s own kind param -- Draw() reads it
76 // to pick which of two tooltip hints to show ("A GfxRender JSON
77 // .scene file" vs. "A single Wavefront .obj file"), since the two
78 // roles this one class fills read differently there even though the
79 // field/button themselves behave identically either way.
80 Kind kind_ = kScene;
81 char scene_path_[256] = "primitives.scene";
82 bool load_requested_ = false;
83};
