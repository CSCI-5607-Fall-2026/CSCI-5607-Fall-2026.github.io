11// Given from Camera & Shading (4) on: plain data in, plain data out (not
12// a callback/visitor interface, and not something that builds Mesh
13// objects itself) -- a consuming app loops over SceneDesc::objects
14// itself and wires each one up to its
15// own scene representation -- for iModel (5) on, that's the actual "hook
16// the loading routine up to your own mesh objects" exercise that
17// assignment wants, kept the same call-and-return shape as Mesh's own
18// factory methods rather than introducing inversion-of-control elsewhere
19// in the given API.
20//
21// Deliberately flat, not a scene graph: SceneDesc::objects is a plain
22// list, each with its own independent world-space transform, not a
23// parent/child hierarchy. A hierarchy could be added later purely
24// additively (an optional "parent" reference per object, or a nested
25// "children" array once the file format is JSON, as it is here) without
26// breaking this shape -- not precluded, just not built until an
27// assignment actually needs it.
28//
29// The on-disk format is JSON (via the vendored nlohmann/json library --
30// see this library's CMakeLists.txt), with every value named rather
31// than positional, so a hand-edited .scene file can't silently
32// mis-order its own fields. See Read()/Write() in scene_io.cc for the
33// exact schema, or a saved .scene file (e.g.
34// gfx_libs/gfxrender/data/teapots.scene) for a concrete example.

75struct SceneMaterialDesc {
76 Color diffuse = Color(0.7f, 0.7f, 0.7f);
77 Color ambient = Color(0.15f, 0.15f, 0.15f);
78 Color specular = Color(0.5f, 0.5f, 0.5f);
79 float specular_exponent = 32.0f;
80 float reflectivity = 0.0f;
81 // Index of refraction (Ni, in Wavefront .mtl terms) and transmission
82 // (Tr) -- how much of a ray passing this material's surface bends
83 // (Snell's law) and how much of the local shading result gets
84 // replaced by that bent, recursively-traced ray, the same 0..1
85 // blend-weight role reflectivity plays for mirror reflection. Not
86 // read by any given shader/rasterizer here -- only Ray (6)'s own
87 // raytracer (see RayTracer::TraceRay()) does anything with them, the
88 // same way reflectivity itself is inert outside that one assignment.
89 // Both default to values that make a material behave as if it had no
90 // refractive properties at all (transmission = 0, so
91 // index_of_refraction is never actually read), so an object with no
92 // refraction of its own round-trips through an older .scene file
93 // exactly as before.
94 float index_of_refraction = 1.0f;
95 float transmission = 0.0f;
96 std::string texture; // bare filename; empty = none
97 // Defaults to kPhong regardless of what else this material carries --
98 // deliberately not tied to any particular scene or consuming app's
99 // own idea of a "default style," so a .scene file saved before this
100 // field existed (or one that simply omits it) always reads back as
101 // Phong, the shading style every given asgn_solns app has already
102 // used as its own baseline default.
103 ShaderStyle shader_style = ShaderStyle::kPhong;
104};

155struct SceneDesc {

157 // The window's own clear color, behind everything else in the
158 // scene -- a consuming app applies this via GraphicsApp::
159 // SetBackgroundColor(). Defaults to the dark blue-gray every given
160 // asgn_solns app has always hardcoded via glClearColor(), not
161 // Color()'s own black default, so an older .scene file with no
162 // "background_color" key at all reads as exactly that same
163 // familiar background rather than going black.
164 Color background_color = Color(0.1f, 0.1f, 0.15f);
165 std::vector<Light> lights;
166 std::vector<SceneObjectDesc> objects;
167};
